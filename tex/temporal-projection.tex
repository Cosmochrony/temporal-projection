  \begin{abstract}
    The note~\cite{Beau2026n} identifies the construction of a temporal residual
    map~$\mathcal{P}_\tau^{\mathrm{Q5b}}$ as the remaining step needed to close the Lorentz
    identification from O7 observables.
    The present paper carries out this construction and proves that the map is the unique
    positive function compatible with three inputs simultaneously:
    the Lorentzian temporal splitting induced by the completed effective co-metric
    $g^{\mu\nu} = 2\eta^{\mu\nu}$ of Q5b--Q11~\cite{Beau2026q5b,Beau2026q11},
    the Born--Infeld unit capacity-flow sphere of~\cite{Beau2026c}, and
    the exact Weil linear filling law $\Sigma_n^{\mathrm{tot}} = 1 - B_n$
    of~\cite{Beau2026n}.

    The completed principal symbol determines the Lorentzian temporal splitting of the
    effective tangent bundle; it does not by itself define a non-linear action on the
    scalar observable~$\Sigma_n^{\mathrm{tot}}$.
    The Born--Infeld admissibility budget provides the canonical unit capacity-flow lift,
    and the Weil filling law identifies the spatial component $B_n$.
    Their composition defines the scalar temporal residual map
    \[
      \mathcal{P}_\tau^{\mathrm{Q5b}} \colon \Sigma_n^{\mathrm{tot}}
      \;\longmapsto\; \sqrt{\Sigma_n^{\mathrm{tot}}\bigl(2 - \Sigma_n^{\mathrm{tot}}\bigr)}
      \;=\; \sqrt{1 - B_n^2}.
    \]
    This map is not an idempotent projection on scalar observables; it is the scalar shadow
    of a linear Lorentzian projection composed with the Born--Infeld lift.
    The three inputs are individually necessary: removing any one of them leaves the
    formula undetermined.
    This necessity retroactively explains the numerical failure of all three candidate
    maps tested in~\cite{Beau2026n}.
    Lemma~1 of~\cite{Beau2026n} follows, and with it Theorem~1:
    $\Phi(\eta_n^\tau) = (1+(\eta_n^\tau)^2)^{-1/2} \to 1/\gamma$.
  \end{abstract}

  \section{Capacity-Flow Bundle and Born--Infeld Sphere}\label{sec:bundle}

    \subsection{The capacity flow vector}

      In the large-$q$ continuum limit, the cascade depth $n$ is identified with the coordinate
      $\tau$ on $\mathbb{R}_\tau$ (Q5b Remark~3.3~\cite{Beau2026q5b}, Q11 Lemma~3.1~\cite{Beau2026q11}).
      The O7 observables become sections of a bundle over $\mathbb{R}_\tau \times \mathrm{Heis}_3(\mathbb{R})$.

      \begin{definition}[Capacity flow vector and unit BI sphere]\label{def:BIsphere}
        The \emph{capacity flow vector} at depth $n$ is
        \[
          \vec{F}_n := F_n^\tau\,\partial_\tau + \vec{F}_n^{\,\mathrm{spat}}
          \;\in\; T_{(\tau,e)}(\mathbb{R}_\tau \times \mathrm{Heis}_3(\mathbb{R})),
        \]
        where $F_n^\tau = dI_\tau/dn$ and $|\vec{F}_n^{\,\mathrm{spat}}| = dI_{\mathrm{spat}}/dn$
        are the normalised temporal and spatial capacity increments.
        The BI budget~\eqref{eq:BI} places $\vec{F}_n$ on the \emph{unit Born--Infeld sphere}:
        \begin{equation}\label{eq:BIsphere}
          (F_n^\tau)^2 + |\vec{F}_n^{\,\mathrm{spat}}|^2 = 1.
        \end{equation}
      \end{definition}

    \subsection{Identification of the spatial component}

      The O7 observable $B_n$ is the normalised spatial coverage of the BFS cascade.
      By the Q5b Carnot--Carath\'eodory convergence (Theorem~3.2 of~\cite{Beau2026q5b}),
      in the large-$q$ limit $B_n$ coincides with the fraction of spatial capacity occupied by
      the BFS ball, i.e.\ the magnitude of the spatial component of the normalised capacity
      flow:
      \begin{equation}\label{eq:spatial}
        B_n = |\vec{F}_n^{\,\mathrm{spat}}|.
      \end{equation}
      Substituting~\eqref{eq:spatial} into the unit BI sphere~\eqref{eq:BIsphere} gives
      \begin{equation}\label{eq:Ftau}
        F_n^\tau = \sqrt{1 - B_n^2}.
      \end{equation}

  \section{Construction and Uniqueness of $\mathcal{P}_\tau^{\mathrm{Q5b}}$}

    \subsection{The linear Lorentzian projector on vectors}

      The completed co-metric~\eqref{eq:metric} endows $\mathbb{R}_\tau \times \mathrm{Heis}_3(\mathbb{R})$
      with a Lorentzian structure.
      The associated \emph{temporal projector} on tangent vectors is the standard Lorentzian
      orthogonal projection onto the timelike direction $\partial_\tau$:
      \begin{equation}\label{eq:Ptau-vector}
        P_\tau V := \frac{g(V,\partial_\tau)}{g(\partial_\tau,\partial_\tau)}\,\partial_\tau
        = \frac{g_{\tau\mu}V^\mu}{g_{\tau\tau}}\,\partial_\tau
        = V^\tau\,\partial_\tau,
      \end{equation}
      using $g_{\tau\tau} = -1/2$ from~\eqref{eq:invmetric} and $g_{\tau\mu}V^\mu = -V^\tau/2$.
      The map $V \mapsto P_\tau V = V^\tau\partial_\tau$ is \emph{linear and idempotent}
      on $T(\mathbb{R}_\tau \times \mathrm{Heis}_3(\mathbb{R}))$.

      \begin{remark}[Scope of the linear projector]\label{rem:scope}
        The principal symbol determines $P_\tau$ via the co-metric~\eqref{eq:metric}.
        It does not, however, determine a map on the scalar observable
        $\Sigma_n^{\mathrm{tot}} \in [0,1]$: $P_\tau$ acts on tangent vectors, not on
        real-valued capacity sections.
        The lift from scalars to vectors is provided by the BI bundle structure
        (Definition~\ref{def:BIsphere}), not by the principal symbol.
      \end{remark}

      \begin{definition}[Temporal residual map]\label{def:Ptau}

        The name ``temporal residual map'' is preferred over ``projector'' for

        orthogonal projector.

    \subsection{Main theorem: explicit formula and uniqueness}

      \begin{theorem}[BI-lifted temporal residual map]\label{thm:main}
        Assume the Q5a continuum hypotheses~\cite{Beau2026q5a}
        \textup{[H1]}, \textup{[H-w]}, \textup{[H-E1]}, \textup{[C]},
        the spectral universality hypothesis~\textup{[U]} proved in U1~\cite{Beau2026u1},
        and the canonical BI capacity-flow lift of Definition~\ref{def:BIsphere}.
        The temporal residual map $\mathcal{P}_\tau^{\mathrm{Q5b}}$ acts on the O7 total
        residual~$\Sigma_n^{\mathrm{tot}} \in [0,1]$ by
        \begin{equation}\label{eq:Ptau-scalar}
          \mathcal{P}_\tau^{\mathrm{Q5b}}\,\Sigma_n^{\mathrm{tot}}
          \;=\; \sqrt{\Sigma_n^{\mathrm{tot}}\bigl(2-\Sigma_n^{\mathrm{tot}}\bigr)}
          \;=\; \sqrt{1-B_n^2}.
        \end{equation}
        Consequently, the quadratic law
        \begin{equation}\label{eq:quadratic}
          B_n^2 \;+\; \bigl(\mathcal{P}_\tau^{\mathrm{Q5b}}\,\Sigma_n^{\mathrm{tot}}\bigr)^2 \;=\; 1
        \end{equation}
        holds exactly for all $n < q/2$.

        Moreover, $\mathcal{P}_\tau^{\mathrm{Q5b}}$ is the \emph{unique} map
        $\psi\colon[0,1]\to[0,1]$ satisfying simultaneously:
        \begin{itemize}
          \item[\textup{(L)}] $\psi(\Sigma_n^{\mathrm{tot}}) = |P_\tau \vec{F}_n|$ (Lorentzian
          projection of the capacity flow);
          \item[\textup{(BI)}] $B_n^2 + \psi(\Sigma_n^{\mathrm{tot}})^2 = 1$
          (unit BI sphere~\eqref{eq:BIsphere});
          \item[\textup{(W)}] $\Sigma_n^{\mathrm{tot}} = 1 - B_n$ (Weil filling law~\eqref{eq:Sigma}).
        \end{itemize}
      \end{theorem}

      \begin{proof}
        \textbf{Explicit formula.}
        By the Weil filling law~\eqref{eq:Sigma}: $B_n = 1 - \Sigma_n^{\mathrm{tot}}$.
        By the spatial identification~\eqref{eq:spatial} and the unit BI sphere~\eqref{eq:BIsphere}:
        \[
          (F_n^\tau)^2 = 1 - B_n^2 = 1 - (1-\Sigma_n^{\mathrm{tot}})^2
          = \Sigma_n^{\mathrm{tot}}(2 - \Sigma_n^{\mathrm{tot}}).
        \]
        By Definition~\ref{def:Ptau}: $\mathcal{P}_\tau^{\mathrm{Q5b}}\,\Sigma_n^{\mathrm{tot}}
        = |F_n^\tau| = \sqrt{\Sigma_n^{\mathrm{tot}}(2-\Sigma_n^{\mathrm{tot}})}$.
        Equation~\eqref{eq:quadratic} follows immediately: $B_n^2 + (1-B_n^2) = 1$.
        This identity is exact for all $n < q/2$.

        \textbf{Uniqueness.}
        Let $\psi\colon[0,1]\to[0,1]$ satisfy (L), (BI), and (W) simultaneously.
        By (W): $B_n = 1 - f$ where $f = \Sigma_n^{\mathrm{tot}}$.
        Substituting into (BI): $(1-f)^2 + \psi(f)^2 = 1$, hence
        $\psi(f)^2 = 1 - (1-f)^2 = f(2-f)$.
        The unique positive branch is $\psi(f) = \sqrt{f(2-f)}$.
        No other positive function satisfies all three conditions.
      \end{proof}

      \begin{remark}[Derivation chain]\label{rem:chain}
        The three conditions (L), (BI), (W) are jointly necessary and sufficient.
        The Lorentzian splitting (L) specifies \emph{which} component of $\vec{F}_n$ is temporal
        (the $\partial_\tau$ direction, with negative $g$-norm).
        The BI sphere (BI) fixes the normalization of $\vec{F}_n$ to unity,
        forcing $(F_n^\tau)^2 = 1 - B_n^2$ once $B_n$ is known.
        The Weil filling law (W) provides the identification $B_n = 1 - \Sigma_n^{\mathrm{tot}}$,
        converting the formula to a function of the observable alone.
        Dropping any one of these three leaves the formula undetermined.
      \end{remark}

  \section{Necessity of Each Ingredient}\label{sec:necessity}

    \subsection{What fails without each ingredient}

      \begin{proposition}[Necessity]\label{prop:necessity}
        Each of conditions \textup{(L)}, \textup{(BI)}, \textup{(W)} is individually
        necessary for the uniqueness in Theorem~\ref{thm:main}.
        \begin{itemize}
          \item Without \textup{(L)}: the temporal component of $\vec{F}_n$ is unspecified;
          any component of $\vec{F}_n$ could play the role of $\Sigma_n^\tau$.
          \item Without \textup{(BI)}: the capacity flow vector is not constrained to the unit
          sphere; $(F_n^\tau)^2 + B_n^2$ can take any value in $(0,\infty)$, and the quadratic
          law~\eqref{eq:quadratic} need not hold.
          \item Without \textup{(W)}: the spatial component $B_n$ is not identified with
          $1-\Sigma_n^{\mathrm{tot}}$; the formula for $F_n^\tau$ as a function of
          $\Sigma_n^{\mathrm{tot}}$ alone is then inaccessible.
        \end{itemize}
      \end{proposition}

      \begin{proof}
        Direct: each condition is used exactly once in the proof of Theorem~\ref{thm:main}.
        Without (L), the map $V \mapsto P_\tau V$ is undefined, so Definition~\ref{def:Ptau}
        has no content.
        Without (BI), equation~\eqref{eq:BIsphere} does not hold, and $|F_n^\tau|$ cannot be
        expressed in terms of $B_n$ alone.
        Without (W), the substitution $B_n = 1 - \Sigma_n^{\mathrm{tot}}$ is unavailable,
        and $\psi$ cannot be expressed as a function of $\Sigma_n^{\mathrm{tot}}$ alone.
      \end{proof}

    \subsection{Retroactive explanation of the candidate failures}\label{sec:candidates}

      The note~\cite{Beau2026n} (Section~6) tests three candidate maps
      $\Sigma_n^A$, $\Sigma_n^B$, $\Sigma_n^C$ and finds that all fail the quadratic
      law $B_n^2 + (\Sigma_n^X)^2 \to 1$ with mean deviations of $45\%$, $13\%$, and $7\%$
      respectively.
      Proposition~\ref{prop:necessity} now gives the structural reason.

      \begin{proposition}[Failure modes]\label{prop:failure}
        The three candidate maps of~\cite{Beau2026n} fail the quadratic law because each
        violates exactly one of conditions~\textup{(L)}, \textup{(BI)}, \textup{(W)}.
        \begin{enumerate}
          \item $\Sigma_n^A = 1 - |{c : \exists(a,b),\,(a,b,c)\in\bigcup_{k\leq n} S_k}|/q$
          (central-coordinate coverage): fails \textup{(W)}.
          The central variable $c$ is not the complement of $B_n$; it saturates at depth
          $n\approx q/2$ while $B_n \approx 1/2$, so $\Sigma_n^A \neq 1 - B_n$.
          This observable measures central BFS coverage, a distinct quantity from the total
          Weil residual.
          \item $\Sigma_n^B = \sqrt{1-B_n}$ (square-root complement): fails \textup{(BI)}.
          It satisfies $B_n + (\Sigma_n^B)^2 = 1$ but not $B_n^2 + (\Sigma_n^B)^2 = 1$.
          This is the identity arising from a linear budget $B_n + \Sigma = 1$ with $\Sigma$
          quadratically normalized, not from the BI sphere.
          \item $\Sigma_n^C = \sqrt{1-\beta_n^2}$ with $\beta_n = n/n_{\max}$ (depth proxy):
          fails \textup{(W)}.
          The normalised depth $\beta_n = n/n_{\max}$ is not equal to $B_n = (2n^2+2n+1)/q^2$
          (they differ by a quadratic-vs-linear discrepancy in $n$), so the Weil identification
          $B_n = 1 - \Sigma_n^{\mathrm{tot}}$ is not satisfied for $\Sigma_n^C$.
        \end{enumerate}
      \end{proposition}

      \begin{proof}
        For $\Sigma_n^A$: the Weil factorization proves $\Sigma_n^{\mathrm{tot}} = 1 - B_n$
        exactly (\cite{Beau2026n} Corollary~1), but $\Sigma_n^A$ measures the distinct
        observable of central-coordinate BFS coverage, which is not equal to $1-B_n$ in the
        mid-range of the cascade.
        For $\Sigma_n^B$: $B_n + (\sqrt{1-B_n})^2 = B_n + 1 - B_n = 1$, confirming it satisfies
        the \emph{linear} complement law, not the BI sphere $B_n^2 + (\Sigma_n^B)^2 = 1$.
        For $\Sigma_n^C$: $\beta_n = n/n_{\max}$ while $B_n = (2n^2+2n+1)/q^2$; these satisfy
        $B_n \approx \beta_n^2 \cdot n_{\max}^2/q^2$, a different quadratic relationship, so
        (W) fails for $\Sigma_n^C$.
      \end{proof}

      \begin{remark}[Informational content]\label{rem:info}
        The failure of $\Sigma_n^C$ by only $7\%$ is now explained: $\beta_n \approx B_n^{1/2}$
        at intermediate depths (since $B_n \propto n^2/q^2$), so $\Sigma_n^C = \sqrt{1-\beta_n^2}
        \approx \sqrt{1-B_n}$, which is close to $\sqrt{1-B_n^2}$ when $B_n$ is small but
        diverges for $B_n \to 1$, consistent with the reported residuals near $\beta_n \approx 0.8$.
      \end{remark}

    \begin{corollary}[Lorentz factor]\label{cor:lorentz}
      Under the hypotheses of Theorem~\ref{thm:main}, the temporally projected saturation ratio
      \[
        \eta_n^\tau := \frac{B_n}{\mathcal{P}_\tau^{\mathrm{Q5b}}\,\Sigma_n^{\mathrm{tot}}}

      \[
        \Phi(\eta_n^\tau) := \bigl(1 + (\eta_n^\tau)^2\bigr)^{-1/2} = \frac{1}{\gamma}.

    \end{corollary}

    \begin{proof}
      Direct from $\mathcal{P}_\tau^{\mathrm{Q5b}}\,\Sigma_n^{\mathrm{tot}} = \sqrt{1-B_n^2}$
      and the definition of $\gamma$.
    \end{proof}
